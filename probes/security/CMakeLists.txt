if(MSQUIC_LIBRARY)
    add_library(identity_probe_buffers STATIC ../../src/cluster/send_queue.cpp ../../src/cluster/stream_decoder.cpp)
    target_include_directories(identity_probe_buffers PUBLIC ../../src/cluster/include)
    rhythm_project_target(identity_probe_buffers)
    add_library(cluster_send_queue ALIAS identity_probe_buffers)
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../src/cluster_transport" cluster_transport)
    add_executable(transport_contract_tests transport_contract_tests.cpp transport_lan.cpp)
    target_link_libraries(transport_contract_tests PRIVATE cluster_transport_candidate)
    rhythm_project_target(transport_contract_tests)
    list(APPEND security_probes transport_contract_tests)
    add_executable(identity_quic_tests ../quic/msquic_probe.cpp ../quic/stress.cpp ../quic/metrics.cpp ../quic/flow_control.cpp room_wire.cpp)
    target_include_directories(identity_quic_tests PRIVATE ../../src/security "${CMAKE_CURRENT_SOURCE_DIR}" ../quic)
    target_include_directories(identity_quic_tests SYSTEM PRIVATE "${MSQUIC_SOURCE}/src/inc")
    target_compile_definitions(identity_quic_tests PRIVATE RHYTHM_IDENTITY_PROBE RHYTHM_QUIC_OPENSSL)
    target_link_libraries(identity_quic_tests PRIVATE cluster_auth identity_probe_buffers "${MSQUIC_LIBRARY}")
    if(WIN32)

    rhythm_project_target(identity_quic_tests)
    list(APPEND security_probes identity_quic_tests)

if(WIN32)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    foreach(probe IN LISTS security_probes)
        add_custom_command(TARGET ${probe} POST_BUILD
            COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../tools/deploy-security-probe.py"
                "$<TARGET_FILE:${probe}>" "${CMAKE_CXX_COMPILER}" "${MSQUIC_RUNTIME}" "${RHYTHM_SECURITY_OPENSSL_ROOT}"
            VERBATIM)
        add_custom_target(deploy_${probe} ALL
            COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../tools/deploy-security-probe.py"
                "$<TARGET_FILE:${probe}>" "${CMAKE_CXX_COMPILER}" "${MSQUIC_RUNTIME}" "${RHYTHM_SECURITY_OPENSSL_ROOT}"
            DEPENDS ${probe}
            VERBATIM)
    endforeach()
endif()
