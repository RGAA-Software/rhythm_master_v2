cmake_minimum_required(VERSION 3.25)
project(RhythmQrReaderProbe LANGUAGES C CXX)
if(MSVC AND DEFINED RHYTHM_MSVC_INCLUDE_PREFIX)
    set(CMAKE_CL_SHOWINCLUDES_PREFIX "${RHYTHM_MSVC_INCLUDE_PREFIX}")
    add_compile_options(/FC)
endif()
include("${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/ProjectOptions.cmake")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Static QR candidate" FORCE)
set(ZXING_READERS ON CACHE BOOL "Decoder only" FORCE)
set(ZXING_WRITERS OFF CACHE STRING "Use the existing Nayuki encoder" FORCE)
set(ZXING_EXAMPLES OFF CACHE BOOL "No upstream applications" FORCE)
set(ZXING_BLACKBOX_TESTS OFF CACHE BOOL "No fetched tests" FORCE)
set(ZXING_UNIT_TESTS OFF CACHE BOOL "No fetched tests" FORCE)
set(ZXING_DEPENDENCIES LOCAL CACHE STRING "No implicit downloads" FORCE)
add_subdirectory(../../third_party/sources/zxing-probe zxing SYSTEM)
add_subdirectory(../../src/foundation foundation)
add_subdirectory(../../src/qr_reader reader)
add_library(qr_fixture_encoder STATIC ../../third_party/sources/qrcodegen/qrcodegen.cpp ../../src/qr/qr_generator.cpp)
target_include_directories(qr_fixture_encoder PUBLIC ../../src/qr/include)
target_include_directories(qr_fixture_encoder SYSTEM PRIVATE ../../third_party/sources/qrcodegen)
target_compile_features(qr_fixture_encoder PRIVATE cxx_std_20)
add_executable(qr_reader_tests qr_reader_tests.cpp)
target_include_directories(qr_reader_tests SYSTEM PRIVATE ../../third_party/sources/qrcodegen)
target_link_libraries(qr_reader_tests PRIVATE qr_reader qr_fixture_encoder)
rhythm_project_target(qr_reader_tests)
if(WIN32)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_custom_command(TARGET qr_reader_tests POST_BUILD
        COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../tools/deploy-qr-reader-probe.py"
            "$<TARGET_FILE:qr_reader_tests>" "${CMAKE_CXX_COMPILER}" VERBATIM)
    add_custom_target(deploy_qr_reader ALL
        COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../tools/deploy-qr-reader-probe.py"
            "$<TARGET_FILE:qr_reader_tests>" "${CMAKE_CXX_COMPILER}"
        DEPENDS qr_reader_tests VERBATIM)
endif()
