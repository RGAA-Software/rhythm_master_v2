cmake_minimum_required(VERSION 3.25)
project(RhythmMaster VERSION 0.1.0 LANGUAGES C CXX)
if(MSVC AND DEFINED RHYTHM_MSVC_INCLUDE_PREFIX)
    set(CMAKE_CL_SHOWINCLUDES_PREFIX "${RHYTHM_MSVC_INCLUDE_PREFIX}")
    add_compile_options(/utf-8)
endif()
include(CTest)
include(cmake/ProjectOptions.cmake)
if(BUILD_TESTING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME source_boundaries COMMAND ${Python3_EXECUTABLE}
        "${PROJECT_SOURCE_DIR}/tools/check-boundaries.py")
    add_test(NAME boundary_lexer COMMAND ${Python3_EXECUTABLE}
        "${PROJECT_SOURCE_DIR}/tools/test-boundary-lexer.py")
endif()
option(RHYTHM_BUILD_WINDOWS_SPIKE "Build the provisional Windows architecture spike" OFF)
option(RHYTHM_RENDER_ONLY "Build only the standalone Null renderer" OFF)
option(RHYTHM_BUILD_PROJECT_IO "Build the Protobuf project codec and transaction tests" OFF)
option(RHYTHM_BUILD_ANDROID_PLAYER "Build the Android Player native library" OFF)
option(RHYTHM_BUILD_MEDIA "Use the validated vcpkg FFmpeg local media profile" OFF)
add_subdirectory(src/rhythm_render)
if(NOT RHYTHM_RENDER_ONLY)
    add_library(asset_types INTERFACE)
    target_include_directories(asset_types INTERFACE "${PROJECT_SOURCE_DIR}/src/assets/include")
    add_library(storage_types INTERFACE)
    target_include_directories(storage_types INTERFACE "${PROJECT_SOURCE_DIR}/src/storage/include")
    add_subdirectory(src/media_types)
    add_library(runtime_inputs INTERFACE)
    target_include_directories(runtime_inputs INTERFACE "${PROJECT_SOURCE_DIR}/src/runtime/include")
    add_subdirectory(src/parameters)
    add_subdirectory(src/performance)
    add_subdirectory(src/media_arrangement)
    add_subdirectory(src/particles)
    add_subdirectory(src/scene3d)
    add_subdirectory(src/geometry2d)
    add_subdirectory(src/image_shader)
    add_subdirectory(src/text_layout)
    include(cmake/SceneDependencies.cmake)
    add_subdirectory(src/model_import)
    include(cmake/PhysicsDependencies.cmake)
    add_subdirectory(src/physics2d)
    add_subdirectory(src/audio_analysis)
    target_link_libraries(runtime_inputs INTERFACE audio_types parameters)
    add_subdirectory(src/foundation)
    add_subdirectory(src/cluster)
    add_subdirectory(src/graph)
    add_subdirectory(src/runtime)
    add_subdirectory(src/editor_application)
endif()
if(RHYTHM_BUILD_ANDROID_PLAYER)
    if(NOT ANDROID OR NOT RHYTHM_BUILD_PROJECT_IO)
        message(FATAL_ERROR "Android Player requires the Android toolchain and runtime package")
    endif()
    include(cmake/AndroidPlayerDependencies.cmake)
    add_subdirectory(src/android_player)
endif()
if(RHYTHM_BUILD_PROJECT_IO OR RHYTHM_BUILD_WINDOWS_SPIKE)
    if(RHYTHM_RENDER_ONLY)
        message(FATAL_ERROR "Project IO requires the graph module")
    endif()
    include(cmake/ProjectIoDependencies.cmake)
    add_subdirectory(src/storage)
    add_subdirectory(src/assets)
    add_subdirectory(src/qr)
    add_subdirectory(src/project_io)
    add_subdirectory(src/model_assets)
    if(RHYTHM_BUILD_MEDIA)
        add_subdirectory(src/media)
        add_subdirectory(src/video_playback)
    endif()
    add_subdirectory(src/prepared_assets)
    add_subdirectory(src/video_sources)
    if(RHYTHM_BUILD_MEDIA)
        add_subdirectory(src/export_core)
    endif()
    add_subdirectory(src/player_core)
    add_subdirectory(src/cluster_player)
    add_subdirectory(src/content)
endif()
if(RHYTHM_BUILD_WINDOWS_SPIKE)
    if(NOT WIN32 OR RHYTHM_RENDER_ONLY)
        message(FATAL_ERROR "The host spike requires Windows and the graph module")
    endif()
    include(cmake/SpikeDependencies.cmake)
    add_subdirectory(src/windows_spike)
    add_subdirectory(src/windows_player)
endif()
if(TARGET audio_playback)
    add_subdirectory(src/player_audio)
endif()
