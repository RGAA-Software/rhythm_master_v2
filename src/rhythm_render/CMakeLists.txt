cmake_minimum_required(VERSION 3.25)
if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
    project(RhythmRender LANGUAGES CXX)
    if(MSVC AND DEFINED RHYTHM_MSVC_INCLUDE_PREFIX)
        set(CMAKE_CL_SHOWINCLUDES_PREFIX "${RHYTHM_MSVC_INCLUDE_PREFIX}")
    endif()
    include(CTest)
endif()
add_library(rhythm_render STATIC src/renderer.cpp src/mesh.cpp src/mesh_store.cpp src/null_backend.cpp src/resource_table.cpp src/layout.cpp)
add_library(Rhythm::Render ALIAS rhythm_render)
target_include_directories(rhythm_render PUBLIC include PRIVATE src)
target_compile_features(rhythm_render PUBLIC cxx_std_20)
if(MSVC)
    target_compile_options(rhythm_render PRIVATE /W4 /WX /utf-8 /permissive-)
else()
    target_compile_options(rhythm_render PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(BUILD_TESTING)
    add_executable(scene_render_contract_tests tests/scene_contract_tests.cpp)
    target_link_libraries(scene_render_contract_tests PRIVATE Rhythm::Render)
    if(MSVC)
        target_compile_options(scene_render_contract_tests PRIVATE /W4 /WX /utf-8 /permissive-)
    else()
        target_compile_options(scene_render_contract_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    add_test(NAME scene_render_contracts COMMAND scene_render_contract_tests)
    add_executable(render_contract_tests tests/render_contract_tests.cpp)
    target_link_libraries(render_contract_tests PRIVATE Rhythm::Render)
    if(MSVC)
        target_compile_options(render_contract_tests PRIVATE /W4 /WX /utf-8 /permissive-)
    else()
        target_compile_options(render_contract_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    add_test(NAME render_contracts COMMAND render_contract_tests)
endif()
