set(qr_source "${PROJECT_SOURCE_DIR}/third_party/sources/qrcodegen")
if(NOT EXISTS "${qr_source}/qrcodegen.cpp")
    message(FATAL_ERROR "Run python tools/prepare-qrcode.py to prepare pinned QR sources")
endif()
add_library(qr_nayuki STATIC "${qr_source}/qrcodegen.cpp")
target_include_directories(qr_nayuki SYSTEM PUBLIC "${qr_source}")
target_compile_features(qr_nayuki PRIVATE cxx_std_20)
add_library(qr_generator STATIC qr_generator.cpp)
target_include_directories(qr_generator PUBLIC include)
target_link_libraries(qr_generator PRIVATE qr_nayuki)
rhythm_project_target(qr_generator)
if(BUILD_TESTING)
    add_executable(qr_contract_tests tests/qr_contract_tests.cpp)
    target_link_libraries(qr_contract_tests PRIVATE qr_generator)
    rhythm_project_target(qr_contract_tests)
    add_test(NAME qr_contracts COMMAND qr_contract_tests "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
endif()
