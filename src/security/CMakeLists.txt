set(RHYTHM_SECURITY_OPENSSL_ROOT "" CACHE PATH "Validated OpenSSL 3.5.8 candidate installation")
option(RHYTHM_SECURITY_SHARED_TLS "Use the isolated shared TLS provider" OFF)
if(NOT EXISTS "${RHYTHM_SECURITY_OPENSSL_ROOT}/include/openssl/opensslv.h")
    message(FATAL_ERROR "Provide the validated OpenSSL candidate; do not silently use another SDK")
endif()
add_library(identity_security STATIC identity.cpp secret_bytes.cpp protected_secret.cpp vault_file.cpp)
target_link_libraries(identity_security PRIVATE atomic_storage)
target_include_directories(identity_security PUBLIC include)
target_include_directories(identity_security SYSTEM PRIVATE "${RHYTHM_SECURITY_OPENSSL_ROOT}/include")
if(WIN32)
    target_link_libraries(identity_security PRIVATE "${RHYTHM_SECURITY_OPENSSL_ROOT}/lib/libcrypto.lib" ws2_32 crypt32 advapi32 user32)
else()
    if(RHYTHM_SECURITY_SHARED_TLS)
        target_link_libraries(identity_security PRIVATE "${RHYTHM_SECURITY_OPENSSL_ROOT}/lib/libcrypto.so" dl m)
    else()
        target_link_libraries(identity_security PRIVATE "${RHYTHM_SECURITY_OPENSSL_ROOT}/lib/libcrypto.a" dl m)
    endif()
endif()
rhythm_project_target(identity_security)
